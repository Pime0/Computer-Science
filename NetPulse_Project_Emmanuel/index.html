# OSI Layer 4 — Transport (TCP) import socket, threading, json class NetPulseTCPServer: def start(self): # AF_INET=IPv4, SOCK_STREAM=TCP self.sock = socket.socket( socket.AF_INET, socket.SOCK_STREAM ) # Avoid TIME_WAIT port conflict self.sock.setsockopt( socket.SOL_SOCKET, socket.SO_REUSEADDR, 1 ) self.sock.bind(('127.0.0.1', 9000)) self.sock.listen(10) # Backlog=10 while True: # Block until TCP handshake done conn, addr = self.sock.accept() # New thread per client t = threading.Thread( target=self.handle, args=(conn, addr) ) t.start() def handle(self, conn, addr): with conn: while True: data = conn.recv(4096) if not data: break conn.sendall(data) # Echo

/* IPv4 Header Structure — 20 bytes min */ struct ip_header { uint8_t version_ihl; /* Version(4) + IHL(4) */ uint8_t tos; /* Type of Service / DSCP */ uint16_t total_length; /* Header + data in bytes */ uint16_t identification; /* Fragmentation ID */ uint16_t flags_fragment; /* DF, MF flags + offset */ uint8_t ttl; /* Time-to-Live (hops) */ uint8_t protocol; /* 6=TCP 17=UDP 1=ICMP */ uint16_t checksum; /* Header integrity check */ uint32_t src_addr; /* Source IP (network order)*/ uint32_t dst_addr; /* Destination IP */ } __attribute__((packed)); /* No padding bytes! */ /* Network byte order conversion */ uint32_t net_ip = htonl(host_ip); /* host→network */ uint16_t net_port = htons(9000); /* little→big endian */ uint32_t host_ip = ntohl(net_ip); /* network→host */

How data travels down the OSI stack
Encapsulation (Sender — Layer 7 → 1)
Decapsulation (Receiver — Layer 1 → 7)
Physical → strips preamble
Data Link → strips Ethernet frame → checks MAC
Network → strips IP header → checks dest IP
Transport → strips TCP/UDP header → routes to port
Application → reads payload data ✓

/* TCP Socket Lifecycle in C */ /* 1. Create socket */ int fd = socket(AF_INET, SOCK_STREAM, 0); /* 2. Set options (avoid ADDRESS IN USE) */ int opt = 1; setsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt)); /* 3. Bind — attach port number */ struct sockaddr_in addr = { .sin_family = AF_INET, .sin_addr.s_addr = INADDR_ANY, .sin_port = htons(9200) /* byte order! */ }; bind(fd, (struct sockaddr*)&addr, sizeof(addr)); /* 4. Listen — mark as passive */ listen(fd, 5); /* backlog = 5 */ /* 5. Accept + thread per client */ while (1) { int client = accept(fd, &cli_addr, &len); pthread_create(&tid, NULL, handle_client, &clients[slot]); pthread_detach(tid); /* auto-cleanup */ }
